ALGORITMA Total Pembayaran

DEKLARASI
  harga_satuan  : real
  jumlah_barang : integer
  biaya_layanan : real
  subtotal      : real
  total         : real

DESKRIPSI
  MULAI
    BACA harga_satuan, jumlah_barang, biaya_layanan
    subtotal <- harga_satuan * jumlah_barang
    total    <- subtotal + biaya_layanan
    TULIS subtotal, total
  SELESAI
